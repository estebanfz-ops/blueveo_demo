Google Sheets Setup — Connect Forms in 5 Steps

Follow these 5 steps — takes about 3 minutes:

  1. Go to sheets.google.com → create a new spreadsheet → name it Kina Studio Data
  2. Rename the first tab to Inquiries. Add two more tabs named Orders and Inventory
  3. Click Extensions → Apps Script. Delete any existing code and paste the script below

function doPost(e) {
  try {
    var d = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();

    if (d.type === 'inquiry') {
      var s = ss.getSheetByName('Inquiries');
      if (s.getLastRow() === 0)
        s.appendRow(['Timestamp','Brand','Contact','Email','Phone',
          'Country','Product Type','Quantity','Budget','Timeline',
          'Contact Method','Details']);
      s.appendRow([new Date(), d.brand, d.contactName, d.email,
        d.phone, d.country, d.productType, d.quantity,
        d.budget, d.timeline, d.contactMethod, d.details]);
    }

    if (d.type === 'order') {
      var o = ss.getSheetByName('Orders');
      if (o.getLastRow() === 0)
        o.appendRow(['Timestamp','Brand','Email','Items','Total COP']);
      o.appendRow([new Date(), d.brand, d.email, d.items, d.total]);

      var inv = ss.getSheetByName('Inventory');
      if (inv.getLastRow() === 0)
        inv.appendRow(['Product','Units Ordered','Logged At']);
      JSON.parse(d.items).forEach(function(item) {
        inv.appendRow([item.name, item.qty, new Date()]);
      });
    }

    return ContentService
      .createTextOutput(JSON.stringify({status:'ok'}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService
      .createTextOutput(JSON.stringify({status:'error',msg:err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
